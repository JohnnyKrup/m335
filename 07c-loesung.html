package ch.wiss.m335.quiz.data;

import androidx.annotation.NonNull;
import androidx.room.ColumnInfo;
import androidx.room.Entity;
import androidx.room.PrimaryKey;

/**
 * Eine Quizfrage als Zeile der Tabelle questions.
 *
 * Vorlage war model/Question. Der Unterschied zu jener Klasse: hier gibt es
 * eine id, die vier Antworten stehen als vier einzelne Felder statt als
 * Array, und es gibt keinen Konstruktor mit Werten. Room baut die Objekte
 * selbst und füllt sie über die Setter.
 *
 * model/Question bleibt daneben bestehen. Das Spiel benutzt sie weiter.
 */
@Entity(tableName = "questions")
public class QuestionEntity {

    /**
     * Die Nummer der Zeile. Room vergibt sie beim Einfügen selbst.
     */
    @PrimaryKey(autoGenerate = true)
    private int id;

    /**
     * Der Fragetext.
     */
    @NonNull
    @ColumnInfo(name = "text")
    private String text = "";

    /**
     * Erste Antwortmöglichkeit.
     */
    @NonNull
    @ColumnInfo(name = "answer1")
    private String answer1 = "";

    /**
     * Zweite Antwortmöglichkeit.
     */
    @NonNull
    @ColumnInfo(name = "answer2")
    private String answer2 = "";

    /**
     * Dritte Antwortmöglichkeit.
     */
    @NonNull
    @ColumnInfo(name = "answer3")
    private String answer3 = "";

    /**
     * Vierte Antwortmöglichkeit.
     */
    @NonNull
    @ColumnInfo(name = "answer4")
    private String answer4 = "";

    /**
     * Die richtige Antwort als Text. Muss einer der vier Antworten entsprechen.
     */
    @NonNull
    @ColumnInfo(name = "correct_answer")
    private String correctAnswer = "";

    /**
     * Kategorie: Sport, Technologie, Geographie oder Geschichte.
     */
    @NonNull
    @ColumnInfo(name = "category")
    private String category = "";

    /**
     * Schwierigkeit: leicht, mittel oder schwer.
     */
    @NonNull
    @ColumnInfo(name = "difficulty")
    private String difficulty = "leicht";

    /**
     * Prüft, ob eine angetippte Antwort richtig ist.
     * <p>
     * Wir vergleichen den Text und nicht die Position. Dadurch bleibt die
     * Prüfung auch dann korrekt, wenn die Antworten gemischt angezeigt werden.
     *
     * @param givenAnswer die Antwort, die der Spieler angetippt hat
     * @return true, wenn die Antwort richtig ist
     */
    public boolean isCorrect(String givenAnswer) {
        return correctAnswer.equals(givenAnswer);
    }

    /**
     * Liefert die vier Antworten als Array.
     *
     * Baut das Array bei jedem Aufruf aus den vier Feldern. Damit sieht diese
     * Klasse von aussen gleich aus wie model/Question.
     *
     * @return die vier Antworten in der Reihenfolge answer1 bis answer4
     */
    public String[] getAnswers() {
        return new String[]{answer1, answer2, answer3, answer4};
    }

    /**
     * Liefert die Nummer der Zeile.
     *
     * @return die id, oder 0 solange die Frage noch nicht gespeichert ist
     */
    public int getId() {
        return id;
    }

    /**
     * Liefert den Fragetext.
     *
     * @return der Fragetext
     */
    public String getText() {
        return text;
    }

    /**
     * Liefert die erste Antwortmöglichkeit.
     *
     * @return die erste Antwort
     */
    public String getAnswer1() {
        return answer1;
    }

    /**
     * Liefert die zweite Antwortmöglichkeit.
     *
     * @return die zweite Antwort
     */
    public String getAnswer2() {
        return answer2;
    }

    /**
     * Liefert die dritte Antwortmöglichkeit.
     *
     * @return die dritte Antwort
     */
    public String getAnswer3() {
        return answer3;
    }

    /**
     * Liefert die vierte Antwortmöglichkeit.
     *
     * @return die vierte Antwort
     */
    public String getAnswer4() {
        return answer4;
    }

    /**
     * Liefert die richtige Antwort als Text.
     *
     * @return die richtige Antwort
     */
    public String getCorrectAnswer() {
        return correctAnswer;
    }

    /**
     * Liefert die Kategorie.
     *
     * @return die Kategorie
     */
    public String getCategory() {
        return category;
    }

    /**
     * Liefert die Schwierigkeit.
     *
     * @return die Schwierigkeit
     */
    public String getDifficulty() {
        return difficulty;
    }

    /**
     * Setzt die Nummer der Zeile.
     *
     * Wird beim Anlegen nicht gebraucht. Room vergibt die id selbst.
     *
     * @param id die Nummer der Zeile
     */
    public void setId(int id) {
        this.id = id;
    }

    /**
     * Setzt den Fragetext.
     *
     * @param text der Fragetext
     */
    public void setText(@NonNull String text) {
        this.text = text;
    }

    /**
     * Setzt die erste Antwortmöglichkeit.
     *
     * @param answer1 die erste Antwort
     */
    public void setAnswer1(@NonNull String answer1) {
        this.answer1 = answer1;
    }

    /**
     * Setzt die zweite Antwortmöglichkeit.
     *
     * @param answer2 die zweite Antwort
     */
    public void setAnswer2(@NonNull String answer2) {
        this.answer2 = answer2;
    }

    /**
     * Setzt die dritte Antwortmöglichkeit.
     *
     * @param answer3 die dritte Antwort
     */
    public void setAnswer3(@NonNull String answer3) {
        this.answer3 = answer3;
    }

    /**
     * Setzt die vierte Antwortmöglichkeit.
     *
     * @param answer4 die vierte Antwort
     */
    public void setAnswer4(@NonNull String answer4) {
        this.answer4 = answer4;
    }

    /**
     * Setzt die richtige Antwort als Text.
     *
     * Der Wert muss einer der vier Antworten entsprechen.
     *
     * @param correctAnswer die richtige Antwort
     */
    public void setCorrectAnswer(@NonNull String correctAnswer) {
        this.correctAnswer = correctAnswer;
    }

    /**
     * Setzt die Kategorie.
     *
     * @param category die Kategorie
     */
    public void setCategory(@NonNull String category) {
        this.category = category;
    }

    /**
     * Setzt die Schwierigkeit.
     *
     * @param difficulty die Schwierigkeit
     */
    public void setDifficulty(@NonNull String difficulty) {
        this.difficulty = difficulty;
    }
}

data/QuestionDao.java

package ch.wiss.m335.quiz.data;

import androidx.room.Dao;
import androidx.room.Delete;
import androidx.room.Insert;
import androidx.room.Query;
import androidx.room.Update;

import java.util.List;

/**
 * Der Zugriff auf die Tabelle questions.
 *
 * Ein Dao beschreibt, was mit der Tabelle erlaubt ist. Es enthält keinen
 * einzigen Methodenrumpf. Die Rümpfe schreibt room-compiler beim Bauen.
 *
 * Die sechs Methoden decken die vier Grundoperationen ab. In 07C läuft
 * davon nur count(), die übrigen werden ab 08A gebraucht.
 */
@Dao
public interface QuestionDao {

    /**
     * Legt eine neue Frage an.
     *
     * Die id im übergebenen Objekt wird nicht beachtet. Room vergibt eine
     * eigene und gibt sie zurück.
     *
     * @param question die neue Frage
     * @return die id, die Room vergeben hat
     */
    @Insert
    long insert(QuestionEntity question);

    /**
     * Liefert alle Fragen, nach Kategorie und dann nach id sortiert.
     *
     * Die Sortierung steht im SQL und nicht in Java. Die Datenbank kann das
     * besser, und die Liste kommt schon fertig geordnet an.
     *
     * @return alle Fragen, nie null
     */
    @Query("SELECT * FROM questions ORDER BY category, id")
    List<QuestionEntity> getAll();

    /**
     * Zählt alle Fragen.
     *
     * Die einzige Methode, die in 07C schon läuft. Sie beantwortet die Frage,
     * ob überhaupt etwas in der Tabelle steht.
     *
     * @return Anzahl Fragen in der Tabelle
     */
    @Query("SELECT COUNT(*) FROM questions")
    int count();

    /**
     * Speichert Änderungen an einer bestehenden Frage.
     *
     * Room erkennt die Frage an ihrer id. Ist die id 0, findet Room keine
     * passende Zeile, und es passiert gar nichts. Das ist der häufigste
     * Fehler beim Bearbeiten.
     *
     * @param question die geänderte Frage, inklusive ihrer id
     */
    @Update
    void update(QuestionEntity question);

    /**
     * Löscht eine Frage.
     *
     * Auch hier zählt nur die id. Die anderen Felder darf Room ignorieren.
     *
     * @param question die zu löschende Frage
     */
    @Delete
    void delete(QuestionEntity question);

    /**
     * Liefert eine einzelne Frage.
     *
     * Wird beim Bearbeiten gebraucht, um das Formular zu füllen.
     *
     * @param id die Nummer der Frage
     * @return die Frage, oder null wenn es sie nicht gibt
     */
    @Query("SELECT * FROM questions WHERE id = :id")
    QuestionEntity findById(int id);
}

data/QuizDatabase.java

package ch.wiss.m335.quiz.data;

import android.content.Context;
import android.util.Log;

import androidx.room.Database;
import androidx.room.Room;
import androidx.room.RoomDatabase;

/**
 * Die lokale Datenbank der App, Dateiname quiz.db.
 *
 * Es gibt sie genau einmal. Das statische Feld instance hält sie fest, und
 * getInstance ist der einzige Weg dorthin. Zwei Datenbanken auf derselben
 * Datei würden sich gegenseitig überschreiben.
 *
 * Die Klasse ist abstract. Room leitet daraus beim Bauen QuizDatabase_Impl
 * ab und schreibt dort die fehlenden Rümpfe hinein.
 */
@Database(entities = {QuestionEntity.class}, version = 1, exportSchema = false)
public abstract class QuizDatabase extends RoomDatabase {
    /**
     * Log-Tag für das ganze Modul 335.
     */
    private static final String TAG = "Modul335";

    /**
     * Der Dateiname der Datenbank auf dem Gerät.
     */
    private static final String DB_NAME = "quiz.db";

    /**
     * Die eine Datenbank. Wird beim ersten Zugriff gebaut.
     */
    private static QuizDatabase instance;

    /**
     * Der Zugriff auf die Tabelle questions.
     * <p>
     * Auch diese Methode ist abstract. Room baut sie selbst.
     *
     * @return das Dao, über das alle Abfragen laufen
     */
    public abstract QuestionDao questionDao();

    /**
     * Liefert die Datenbank und baut sie beim ersten Aufruf.
     *
     * @param context irgendein Context, zum Beispiel eine Activity
     * @return die eine Datenbank der App
     */
    public static synchronized QuizDatabase getInstance(Context context) {
        if (instance == null) {
            Context appContext = context.getApplicationContext();

            instance = Room.databaseBuilder(appContext, QuizDatabase.class, DB_NAME)
                    .allowMainThreadQueries()
                    .build();

            // build() legt die Datei noch nicht an. Das passiert erst beim
            // ersten Zugriff, also bei count() gleich hier darunter.
            Log.d(TAG, "database opened, name=" + DB_NAME
                    + ", questions=" + instance.questionDao().count());
        }
        return instance;
    }
}

MainActivity.java

package ch.wiss.m335.quiz;

import android.content.Intent;
import android.os.Bundle;
import android.util.Log;
import android.widget.Button;
import android.widget.TextView;

import androidx.activity.EdgeToEdge;
import androidx.activity.result.ActivityResultLauncher;
import androidx.activity.result.contract.ActivityResultContracts;
import androidx.annotation.NonNull;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;

import ch.wiss.m335.quiz.data.QuizDatabase;
import ch.wiss.m335.quiz.data.ScoreStorage;

/**
 * Der Startbildschirm der Quiz-App.
 *
 * Vier Knöpfe, dazu die Anzeige der letzten und der besten Punktzahl.
 *
 * Seit 07C öffnet diese Klasse ausserdem die Datenbank. Der Aufruf
 * QuizDatabase.getInstance(this) steht in onCreate und ist die erste
 * Stelle der App, die auf quiz.db zugreift.
 */
public class MainActivity extends AppCompatActivity {

    /**
     * Log-Tag für das ganze Modul 335.
     *
     * Bewusst nicht "M335". Diese Zeichenfolge steckt auch im
     * Paketnamen ch.wiss.m335.quiz. Das Suchfeld im Logcat sucht im
     * Freitext über alle Spalten, und damit träfe "M335" jede Zeile
     * unserer App statt nur die selbst geschriebenen.
     */
    private static final String TAG = "Modul335";

    /** Bedeutet: es wurde noch keine Runde zu Ende gespielt. */
    private static final int NO_ROUND_YET = -1;

    /** Schlüssel für die Punkte der letzten Runde im Bundle. */
    private static final String KEY_LAST_SCORE = "lastScore";

    /** Schlüssel für die Anzahl Fragen der letzten Runde im Bundle. */
    private static final String KEY_LAST_TOTAL = "lastTotal";

    /** Die Anzeige des letzten Punktestands, unter dem Titel. */
    private TextView tvLastScore;

    /** Die Anzeige des besten Werts, direkt unter dem Titel. */
    private TextView tvBestScore;

    /** Die Punkte der letzten beendeten Runde, oder NO_ROUND_YET. */
    private int lastScore = NO_ROUND_YET;

    /** Wie viele Fragen die letzte beendete Runde hatte. */
    private int lastTotal;

    /**
     * Der Kanal, über den die GameActivity ihr Ergebnis zurückgibt.
     *
     * Der Launcher muss in onCreate entstehen und nicht erst beim Klick.
     */
    private ActivityResultLauncher<Intent> gameLauncher;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);

        // 16dp eigener Rand, in Pixel umgerechnet über die Dichte des Geräts.
        final int extraPadding = Math.round(16 * getResources().getDisplayMetrics().density);

        // Wichtig: der eigene Rand wird zu den System-Insets dazugerechnet.
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.clMain), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(
                    systemBars.left + extraPadding,
                    systemBars.top + extraPadding,
                    systemBars.right + extraPadding,
                    systemBars.bottom + extraPadding);
            return insets;
        });

        tvLastScore = findViewById(R.id.tvLastScore);
        tvBestScore = findViewById(R.id.tvBestScore);

        gameLauncher = registerForActivityResult(
                new ActivityResultContracts.StartActivityForResult(),
                result -> {
                    if (result.getResultCode() == RESULT_OK && result.getData() != null) {
                        lastScore = result.getData().getIntExtra(GameActivity.EXTRA_SCORE, 0);
                        lastTotal = result.getData().getIntExtra(GameActivity.EXTRA_TOTAL, 0);
                        showLastScore();
                        Log.d(TAG, "result received, score=" + lastScore + "/" + lastTotal);
                    }
                });

        Button btnPlay = findViewById(R.id.btnPlay);
        Button btnAbout = findViewById(R.id.btnAbout);
        Button btnRanking = findViewById(R.id.btnRanking);

        // Der Neubau bringt seinen Zettel mit. Beim allerersten Start ist er
        // null, dann bleiben die Felder auf ihrem Anfangswert.
        if (savedInstanceState != null) {
            lastScore = savedInstanceState.getInt(KEY_LAST_SCORE, NO_ROUND_YET);
            lastTotal = savedInstanceState.getInt(KEY_LAST_TOTAL, 0);

            Log.d(TAG, "state restored, lastScore=" + lastScore + "/" + lastTotal);
        }

        // Ein Intent nennt zwei Dinge: von wo aus und wohin.
        // Mehr braucht ein Start ohne Rückgabe nicht.
        btnPlay.setOnClickListener(v -> gameLauncher.launch(new Intent(this, GameActivity.class)));
        btnAbout.setOnClickListener(v -> startActivity(new Intent(this, AboutActivity.class)));
        btnRanking.setOnClickListener(v -> startActivity(new Intent(this, HighscoreActivity.class)));

        showLastScore();

        // Öffnet die Datenbank ein einziges Mal beim Start der App. Die
        // Meldung dazu schreibt QuizDatabase selbst.
        QuizDatabase.getInstance(this);

        Log.d(TAG, "MainActivity onCreate");
    }

    /**
     * Läuft jedes Mal, wenn dieser Bildschirm in den Vordergrund kommt.
     *
     * Also nach dem Start, nach der Rückkehr aus der GameActivity und nach
     * dem Wechsel aus einer anderen App.
     */
    @Override
    protected void onResume() {
        super.onResume();
        showBestScore();
    }

    /**
     * Legt die Werte weg, die den Neubau dieser Activity überstehen sollen.
     *
     * @param outState der Zettel, den der Neubau später zurückbekommt
     */
    @Override
    protected void onSaveInstanceState(@NonNull Bundle outState) {
        super.onSaveInstanceState(outState);
        outState.putInt(KEY_LAST_SCORE, lastScore);
        outState.putInt(KEY_LAST_TOTAL, lastTotal);

        Log.d(TAG, "state saved, lastScore=" + lastScore + "/" + lastTotal);
    }

    /**
     * Schreibt den letzten Punktestand in die Anzeige.
     *
     * Es gibt genau eine Stelle, die diesen Text setzt.
     */
    private void showLastScore() {
        if (lastScore == NO_ROUND_YET) {
            tvLastScore.setText(getString(R.string.last_score_none));
        } else {
            tvLastScore.setText(getString(R.string.last_score_format, lastScore, lastTotal));
        }
    }

    /** Schreibt den besten Wert aus SharedPreferences in die Anzeige. */
    private void showBestScore() {
        int best = ScoreStorage.getBestScore(this);
        if (best == ScoreStorage.NO_SCORE_YET) {
            tvBestScore.setText(getString(R.string.best_score_none));
        } else {
            tvBestScore.setText(getString(R.string.best_score_format, best));
        }

        Log.d(TAG, "best score read, value=" + best);
    }
}
