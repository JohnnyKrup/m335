08A Lösung: Startbestand und Fragenliste

Modul 335 Mobile Applikationen, Tag 4 Nachmittag. Lösungsmaterial zum Auftrag 08A. Version 5 vom 06.09.2026.

Unten stehen die neun Dateien vollständig, so wie sie nach 08A aussehen. Die Kommentare darin sind die ausführliche Fassung, im Block hattest du die kürzeren.

Prüfliste

NrDas prüfst duWarum es zählt
1Steht in seedIfEmpty ganz oben die Prüfung auf dao.count() mit einem return?Der wichtigste Punkt. Ohne sie wächst der Bestand bei jedem Start um dreissig, und nichts stürzt dabei ab
2Steht der Aufruf seedIfEmpty(appContext) innerhalb des if (instance == null)?Sonst läuft er bei jedem Zugriff auf die Datenbank. Mit Punkt 1 ist er harmlos, aber er kostet jedes Mal eine Abfrage
3Steht die Zeile database opened nach seedIfEmpty?Darüber zählt sie, bevor etwas eingefügt ist, und meldet für immer null. Beim doppelten Seed meldet sie 60 und ist damit deine zweite Kontrolle
4Steht in deiner Schleife kein setId?Room vergibt die id selbst. Wer die aus der Datei übernimmt, überschreibt in 08B mit der ersten selbst angelegten Frage eine bestehende
5Tragen die vier setAnswer-Zeilen die Zahlen 1, 2, 3 und 4?Vier fast gleiche Zeilen, und keine Meldung, wenn eine Zahl doppelt steht. Die Antworten stehen auf keinem Bildschirm, der Fehler schläft bis 08C
6Steht der Dateiname als Konstante ASSET_NAME an einer Stelle?Der Compiler prüft ihn nicht. An zwei Stellen ist einer davon irgendwann falsch, und niemand meldet es
7Heisst der Ordner genau assets und liegt er unter app/src/main?Android sucht ihn dort und nirgendwo sonst. Ein Ordner an einem anderen Ort wird nicht mitgeliefert
8Steht StandardCharsets.UTF_8 beim Lesen dabei?Ohne die Angabe entscheidet das Gerät. Auf einem läuft es, auf einem anderen stehen Fragezeichen statt Umlaute
9Loggt der catch-Block, statt leer zu sein?Ein leerer catch macht aus einem Fehler eine leere Liste ohne Grund. Das ist die teuerste Art zu scheitern
10Hängen am rvQuestions alle drei, LayoutManager, Adapter und ein Zeilenlayout?Fehlt einer, siehst du eine weisse Fläche. Dieselben drei wie in 06C
11Wird der Zähler in onResume gesetzt, aus derselben Liste wie der Adapter?Zwei Quellen für dieselbe Zahl gehen irgendwann auseinander. Aus einer Liste kann sie gar nicht falsch sein
12Steht das Umschalten in einer Methode showEmptyState?Es ist immer genau eines von beiden zu sehen. Zwei Orte für eine Entscheidung gehen irgendwann auseinander
13Baut der QuestionAdapter keinen Intent und entscheidet er nichts?Er zeigt an und meldet einen Klick weiter. Ein Adapter, der auch entscheidet, lässt sich nirgends wiederverwenden
14Steht im QuestionAdapter nirgends mehr vier, drei oder Rangliste?Alles kommt aus der Vorlage. Ein Kommentar mit einer Zahl darin veraltet als Erstes, und ein falscher Kommentar ist schlimmer als gar keiner
15Haben alle neuen Klassen, Felder und Methoden ihr Javadoc?Wird in der Leistungsbeurteilung bewertet. Nachträglich dauert es zehnmal so lang

Wenn ein Punkt bei dir anders ist und trotzdem funktioniert, entscheide selbst. Aber du solltest sagen können, warum.

package ch.wiss.m335.quiz;

import android.os.Bundle;
import android.util.Log;
import android.view.View;
import android.widget.TextView;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;
import androidx.recyclerview.widget.LinearLayoutManager;
import androidx.recyclerview.widget.RecyclerView;

import ch.wiss.m335.quiz.data.QuestionEntity;
import ch.wiss.m335.quiz.data.QuizDatabase;

import java.util.List;

/**
 * Zeigt die Fragen, die in der Datenbank stehen.
 *
 * Der Bildschirm hat zwei Gesichter. Sind Fragen da, zeigt er die Liste und
 * ihre Anzahl im Titel. Ist die Tabelle leer, zeigt er stattdessen einen
 * Satz, der sagt was los ist.
 *
 * Die Daten kommen über das QuestionDao aus der QuizDatabase. Geladen wird
 * in onResume und nicht in onCreate, damit ab 08B eine frisch angelegte
 * Frage bei der Rückkehr sofort in der Liste steht.
 */
public class AdminActivity extends AppCompatActivity implements QuestionAdapter.OnQuestionClickListener {

    /** Log-Tag für das ganze Modul 335. Derselbe wie in den anderen Activities. */
    private static final String TAG = "Modul335";

    /** Der Titel oben. Er trägt die Anzahl Fragen. */
    private TextView tvAdminTitle;

    /** Der Text, der erscheint, wenn keine Frage da ist. */
    private TextView tvEmpty;

    /** Die Liste der Fragen. */
    private RecyclerView rvQuestions;

    /** Der Adapter, der die Zeilen füllt. */
    private QuestionAdapter adapter;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_admin);

        // 16dp eigener Rand, in Pixel umgerechnet über die Dichte des Geräts.
        final int extraPadding = Math.round(16 * getResources().getDisplayMetrics().density);

        // Wichtig: der eigene Rand wird zu den System-Insets dazugerechnet.
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.clAdmin), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(
                    systemBars.left + extraPadding,
                    systemBars.top + extraPadding,
                    systemBars.right + extraPadding,
                    systemBars.bottom + extraPadding);
            return insets;
        });

        tvAdminTitle = findViewById(R.id.tvAdminTitle);
        tvEmpty = findViewById(R.id.tvEmpty);
        rvQuestions = findViewById(R.id.rvQuestions);

        // Ohne LayoutManager bleibt die Liste leer. Keine Exception, kein
        // Eintrag unter unserem Tag, nur eine leere Fläche.
        rvQuestions.setLayoutManager(new LinearLayoutManager(this));

        adapter = new QuestionAdapter(this);
        rvQuestions.setAdapter(adapter);

        Log.d(TAG, "AdminActivity onCreate");
    }

    /**
     * Reagiert auf einen Klick in der Liste.
     *
     * Heute nur ein Log-Eintrag. In 08C öffnet diese Methode das
     * Formular zum Bearbeiten.
     *
     * @param question die angetippte Frage
     */
    @Override
    public void onQuestionClick(QuestionEntity question) {
        Log.d(TAG, "question clicked, id=" + question.getId());
    }

    /**
     * Schaltet zwischen Liste und leerem Zustand um.
     *
     * Es ist immer genau eines von beiden zu sehen. Darum steht das
     * Umschalten in einer einzigen Methode und nicht an zwei Stellen.
     *
     * @param empty true, wenn keine Frage da ist
     */
    private void showEmptyState(boolean empty) {
        tvEmpty.setVisibility(empty ? View.VISIBLE : View.GONE);
        rvQuestions.setVisibility(empty ? View.GONE : View.VISIBLE);
    }

    /**
     * Lädt die Fragen jedes Mal neu, wenn der Bildschirm erscheint.
     *
     * Dieselbe Überlegung wie bei der Rangliste: onCreate läuft nur
     * einmal, onResume bei jeder Rückkehr. Ab 08B wird das wichtig, denn
     * dann kommt man von einem Formular hierher zurück.
     */
    @Override
    protected void onResume() {
        super.onResume();

        List<QuestionEntity> questions =
                QuizDatabase.getInstance(this).questionDao().getAll();

        adapter.setQuestions(questions);
        tvAdminTitle.setText(getString(R.string.title_admin_format, questions.size()));
        showEmptyState(questions.isEmpty());

        Log.d(TAG, "questions loaded, total=" + questions.size());
    }
}

java/ch/wiss/m335/quiz/QuestionAdapter.java

package ch.wiss.m335.quiz;

import android.content.Context;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.TextView;

import androidx.annotation.NonNull;
import androidx.recyclerview.widget.RecyclerView;

import ch.wiss.m335.quiz.data.QuestionEntity;

import java.util.ArrayList;
import java.util.List;

/**
 * Füllt die Liste der Fragen im Bildschirm der Fragenverwaltung.
 *
 * Der Adapter zeigt an und entscheidet nichts. Einen Klick auf eine Zeile
 * meldet er über OnQuestionClickListener an die Activity weiter.
 *
 * Entstanden aus dem HighscoreAdapter aus 06C und 07B. Der Aufbau ist
 * derselbe, die Zeile hat hier zwei TextViews statt vier.
 */
public class QuestionAdapter extends RecyclerView.Adapter<QuestionAdapter.QuestionViewHolder> {

    /**
     * Die Einträge, die gerade angezeigt werden.
     *
     * Die Liste gehört dem Adapter. Ausgetauscht wird ihr Inhalt nur über
     * setQuestions, damit das notifyDataSetChanged nie vergessen geht.
     */
    private final List<QuestionEntity> questions = new ArrayList<>();

    /**
     * Wird gerufen, wenn jemand eine Zeile antippt.
     *
     * Ein eigenes Interface mit genau einer Methode. Die Activity sagt
     * damit: gib mir Bescheid, den Rest mache ich selbst.
     */
    public interface OnQuestionClickListener {
        void onQuestionClick(QuestionEntity question);
    }

    /** Wer über Klicks informiert werden will. */
    private final OnQuestionClickListener listener;

    /**
     * @param listener die Activity, die auf Klicks reagiert
     */
    public QuestionAdapter(OnQuestionClickListener listener) {
        this.listener = listener;
    }

    /**
     * Hält die Views einer Zeile fest.
     *
     * Das ist der Sinn des ViewHolder: findViewById läuft einmal pro
     * Zeile und nicht bei jedem Scrollen. In 04B gab es das nicht, dort
     * wurde jede TextView neu erzeugt.
     */
    static class QuestionViewHolder extends RecyclerView.ViewHolder {

        /** Der Fragetext, oben in der Zeile. */
        final TextView tvQuestionText;

        /** Kategorie und Schwierigkeit, klein darunter. */
        final TextView tvQuestionMeta;

        /**
         * Sucht die zwei TextViews einmal und merkt sie sich.
         *
         * @param itemView die frisch gebaute Zeile
         */
        QuestionViewHolder(@NonNull View itemView) {
            super(itemView);
            tvQuestionText = itemView.findViewById(R.id.tvQuestionText);
            tvQuestionMeta = itemView.findViewById(R.id.tvQuestionMeta);
        }
    }


    /**
     * Baut eine neue, leere Zeile aus item_question.xml.
     *
     * Das dritte Argument von inflate ist false. Mit true würde die Zeile
     * sofort an den Parent gehängt, und der RecyclerView bekäme eine
     * IllegalStateException mit dem Text "The specified child already has
     * a parent". Das ist der häufigste Absturz dieser Stufe.
     *
     * @param parent   der RecyclerView, der die Zeile später einhängt
     * @param viewType wird hier nicht gebraucht, es gibt nur eine Zeilenart
     * @return ein ViewHolder mit den zwei TextViews darin
     */
    @NonNull
    @Override
    public QuestionAdapter.QuestionViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
        View row = LayoutInflater.from(parent.getContext())
                .inflate(R.layout.item_question, parent, false);
        return new QuestionViewHolder(row);
    }

    /**
     * Füllt eine vorhandene Zeile mit den Daten einer Frage.
     *
     * Eine Frage hat keinen Rang. Ihre Stelle in der Liste bedeutet nichts,
     * darum steht hier kein position + 1 mehr. Was eine Frage hat, ist eine
     * id, und die vergibt die Datenbank.
     *
     * @param holder   die Zeile, die gefüllt wird
     * @param position der Platz in der Liste, beginnend bei 0
     */
    @Override
    public void onBindViewHolder(@NonNull QuestionAdapter.QuestionViewHolder holder, int position) {
        QuestionEntity question = questions.get(position);
        Context context = holder.itemView.getContext();

        holder.tvQuestionText.setText(question.getText());
        holder.tvQuestionMeta.setText(context.getString(R.string.question_meta_format, question.getCategory(), question.getDifficulty()));

        // Der Listener wird bei jedem Binden neu gesetzt, und das ist
        // richtig so. Die Zeile wird beim Scrollen wiederverwendet und
        // zeigt danach eine andere Frage.
        holder.itemView.setOnClickListener(v -> listener.onQuestionClick(question));
    }

    /** @return wie viele Zeilen die Liste hat */
    @Override
    public int getItemCount() {
        return questions.size();
    }


    /**
     * Tauscht den Inhalt der Liste aus und meldet es dem RecyclerView.
     *
     * Der einzige Weg, auf dem neue Daten in den Adapter kommen. Damit kann
     * das notifyDataSetChanged nicht vergessen gehen.
     *
     * @param newQuestions die Fragen, die ab jetzt angezeigt werden
     */
    @SuppressWarnings("NotifyDataSetChanged")
    public void setQuestions(List<QuestionEntity> newQuestions) {
        questions.clear();
        questions.addAll(newQuestions);
        notifyDataSetChanged();
    }
}

     * Freitext über alle Spalten, und damit träfe "M335" jede Zeile
     * unserer App statt nur die selbst geschriebenen.
     */
    private static final String TAG = "Modul335";

    /** Bedeutet: es wurde noch keine Runde zu Ende gespielt. */
    private static final int NO_ROUND_YET = -1;

    /** Schlüssel für die Punkte der letzten Runde im Bundle. */
    private static final String KEY_LAST_SCORE = "lastScore";

    /** Schlüssel für die Anzahl Fragen der letzten Runde im Bundle. */
    private static final String KEY_LAST_TOTAL = "lastTotal";

    /** Die Anzeige des letzten Punktestands, unter dem Titel. */
    private TextView tvLastScore;

    /** Die Anzeige des besten Werts, direkt unter dem Titel. */
    private TextView tvBestScore;

    /** Die Punkte der letzten beendeten Runde, oder NO_ROUND_YET. */
    private int lastScore = NO_ROUND_YET;

    /** Wie viele Fragen die letzte beendete Runde hatte. */
    private int lastTotal;

    /**
     * Der Kanal, über den die GameActivity ihr Ergebnis zurückgibt.
     *
     * Der Launcher muss in onCreate entstehen und nicht erst beim Klick.
     */
    private ActivityResultLauncher<Intent> gameLauncher;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);

        // 16dp eigener Rand, in Pixel umgerechnet über die Dichte des Geräts.
        final int extraPadding = Math.round(16 * getResources().getDisplayMetrics().density);

        // Wichtig: der eigene Rand wird zu den System-Insets dazugerechnet.
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.clMain), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(
                    systemBars.left + extraPadding,
                    systemBars.top + extraPadding,
                    systemBars.right + extraPadding,
                    systemBars.bottom + extraPadding);
            return insets;
        });

        tvLastScore = findViewById(R.id.tvLastScore);
        tvBestScore = findViewById(R.id.tvBestScore);

        gameLauncher = registerForActivityResult(
                new ActivityResultContracts.StartActivityForResult(),
                result -> {
                    if (result.getResultCode() == RESULT_OK && result.getData() != null) {
                        lastScore = result.getData().getIntExtra(GameActivity.EXTRA_SCORE, 0);
                        lastTotal = result.getData().getIntExtra(GameActivity.EXTRA_TOTAL, 0);
                        showLastScore();
                        Log.d(TAG, "result received, score=" + lastScore + "/" + lastTotal);
                    }
                });

        Button btnPlay = findViewById(R.id.btnPlay);
        Button btnAbout = findViewById(R.id.btnAbout);
        Button btnRanking = findViewById(R.id.btnRanking);
        Button btnQuestions = findViewById(R.id.btnQuestions);

        // Der Neubau bringt seinen Zettel mit. Beim allerersten Start ist er
        // null, dann bleiben die Felder auf ihrem Anfangswert.
        if (savedInstanceState != null) {
            lastScore = savedInstanceState.getInt(KEY_LAST_SCORE, NO_ROUND_YET);
            lastTotal = savedInstanceState.getInt(KEY_LAST_TOTAL, 0);

            Log.d(TAG, "state restored, lastScore=" + lastScore + "/" + lastTotal);
        }

        // Ein Intent nennt zwei Dinge: von wo aus und wohin.
        // Mehr braucht ein Start ohne Rückgabe nicht.
        btnPlay.setOnClickListener(v -> gameLauncher.launch(new Intent(this, GameActivity.class)));
        btnAbout.setOnClickListener(v -> startActivity(new Intent(this, AboutActivity.class)));
        btnRanking.setOnClickListener(v -> startActivity(new Intent(this, HighscoreActivity.class)));
        btnQuestions.setOnClickListener(v -> startActivity(new Intent(this, AdminActivity.class)));

        showLastScore();

        // Öffnet die Datenbank ein einziges Mal beim Start der App. Die
        // Meldung dazu schreibt QuizDatabase selbst.
        QuizDatabase.getInstance(this);

        Log.d(TAG, "MainActivity onCreate");
    }

    /**
     * Läuft jedes Mal, wenn dieser Bildschirm in den Vordergrund kommt.
     *
     * Also nach dem Start, nach der Rückkehr aus der GameActivity und nach
     * dem Wechsel aus einer anderen App.
     */
    @Override
    protected void onResume() {
        super.onResume();
        showBestScore();
    }

    /**
     * Legt die Werte weg, die den Neubau dieser Activity überstehen sollen.
     *
     * @param outState der Zettel, den der Neubau später zurückbekommt
     */
    @Override
    protected void onSaveInstanceState(@NonNull Bundle outState) {
        super.onSaveInstanceState(outState);
        outState.putInt(KEY_LAST_SCORE, lastScore);
        outState.putInt(KEY_LAST_TOTAL, lastTotal);

        Log.d(TAG, "state saved, lastScore=" + lastScore + "/" + lastTotal);
    }

    /**
     * Schreibt den letzten Punktestand in die Anzeige.
     *
     * Es gibt genau eine Stelle, die diesen Text setzt.
     */
    private void showLastScore() {
        if (lastScore == NO_ROUND_YET) {
            tvLastScore.setText(getString(R.string.last_score_none));
        } else {
            tvLastScore.setText(getString(R.string.last_score_format, lastScore, lastTotal));
        }
    }

    /** Schreibt den besten Wert aus SharedPreferences in die Anzeige. */
    private void showBestScore() {
        int best = ScoreStorage.getBestScore(this);
        if (best == ScoreStorage.NO_SCORE_YET) {
            tvBestScore.setText(getString(R.string.best_score_none));
        } else {
            tvBestScore.setText(getString(R.string.best_score_format, best));
        }

        Log.d(TAG, "best score read, value=" + best);
    }
}

package ch.wiss.m335.quiz.data;

import android.content.Context;
import android.util.Log;

import androidx.room.Database;
import androidx.room.Room;
import androidx.room.RoomDatabase;

import org.json.JSONArray;
import org.json.JSONException;
import org.json.JSONObject;

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.nio.charset.StandardCharsets;

@Database(entities = {QuestionEntity.class}, version = 1, exportSchema = false)
public abstract class QuizDatabase extends RoomDatabase {
    /**
     * Log-Tag für das ganze Modul 335.
     */
    private static final String TAG = "Modul335";

    /**
     * Der Dateiname der Datenbank auf dem Gerät.
     */
    private static final String DB_NAME = "quiz.db";

    /** Der Startbestand, liegt unter app/src/main/assets. */
    private static final String ASSET_NAME = "questions_m335.json";

    /**
     * Die eine Datenbank. Wird beim ersten Zugriff gebaut.
     */
    private static QuizDatabase instance;

    /**
     * Der Zugriff auf die Tabelle questions.
     * <p>
     * Auch diese Methode ist abstract. Room baut sie selbst.
     *
     * @return das Dao, über das alle Abfragen laufen
     */
    public abstract QuestionDao questionDao();

    /**
     * Liefert die Datenbank und baut sie beim ersten Aufruf.
     *
     * @param context irgendein Context, zum Beispiel eine Activity
     * @return die eine Datenbank der App
     */
    public static synchronized QuizDatabase getInstance(Context context) {
        if (instance == null) {
            Context appContext = context.getApplicationContext();

            instance = Room.databaseBuilder(appContext, QuizDatabase.class, DB_NAME)
                    .allowMainThreadQueries()
                    .build();

            seedIfEmpty(appContext);

            // build() legt die Datei noch nicht an. Das passiert erst beim
            // ersten Zugriff, also bei count() gleich hier darunter.
            Log.d(TAG, "database opened, name=" + DB_NAME
                    + ", questions=" + instance.questionDao().count());
        }
        return instance;
    }

    /**
     * Liest den Startbestand aus dem Ordner assets.
     *
     * assets ist kein res-Ordner. Was dort liegt, wandert unverändert in
     * die App und bekommt keine R-Nummer. Gefunden wird es über den
     * Dateinamen, und darum meldet auch kein Compiler einen Tippfehler
     * darin.
     *
     * Fehlt die Datei, liefert diese Methode ein leeres Array. Die App
     * läuft dann weiter und zeigt den leeren Zustand. Im Logcat steht der
     * Grund.
     *
     * @param appContext der Context der App
     * @return der Inhalt der Datei, oder ein leeres JSON-Array
     */
    private static String readAsset(Context appContext) {
        StringBuilder builder = new StringBuilder();

        try (BufferedReader reader = new BufferedReader(new InputStreamReader(
                appContext.getAssets().open(ASSET_NAME), StandardCharsets.UTF_8))) {

            String line;
            while ((line = reader.readLine()) != null) {
                builder.append(line);
            }

        } catch (IOException e) {
            Log.e(TAG, "asset not found: " + ASSET_NAME, e);
            return "[]";
        }

        return builder.toString();
    }

    /**
     * Füllt die Tabelle mit dem Startbestand aus assets.
     *
     * Läuft einmal beim Aufbau der Datenbank. Steht schon etwas in der
     * Tabelle, steigt die Methode gleich wieder aus und fasst nichts an.
     * Ohne diese Bedingung kämen bei jedem Start der App dreissig Fragen
     * dazu, denn beim Beenden verschwindet das Feld instance.
     *
     * Die id aus der Datei wird nicht übernommen. Room vergibt sie selbst,
     * sonst kollidieren die Fragen aus assets mit selbst angelegten.
     *
     * @param appContext der Context der App, für den Zugriff auf assets
     */
    private static void seedIfEmpty(Context appContext) {
        QuestionDao dao = instance.questionDao();

        int existing = dao.count();
        if (existing > 0) {
            Log.d(TAG, "seed skipped, count=" + existing);
            return;
        }

        try {
            JSONArray array = new JSONArray(readAsset(appContext));

            for(int i = 0; i < array.length(); i++) {
                JSONObject object = array.getJSONObject(i);
                JSONArray answers = object.getJSONArray("answers");

                QuestionEntity question = new QuestionEntity();
                question.setText(object.getString("text"));
                question.setAnswer1(answers.getString(0));
                question.setAnswer2(answers.getString(1));
                question.setAnswer3(answers.getString(2));
                question.setAnswer4(answers.getString(3));
                question.setCorrectAnswer(object.getString("correctAnswer"));
                question.setCategory(object.getString("category"));
                question.setDifficulty(object.getString("difficulty"));

                dao.insert(question);
            }

            Log.d(TAG, "seed done, count=" + dao.count());

        } catch (JSONException e) {
            // Die Datei ist da, aber sie passt nicht zum erwarteten Aufbau.
            Log.e(TAG, "seed failed, json not readable", e);
        }
    }

}
