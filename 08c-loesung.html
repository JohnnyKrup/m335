package ch.wiss.m335.quiz;

import static ch.wiss.m335.quiz.QuestionEditActivity.EXTRA_QUESTION_ID;

import android.content.Intent;
import android.os.Bundle;
import android.util.Log;
import android.view.View;
import android.widget.TextView;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;
import androidx.recyclerview.widget.LinearLayoutManager;
import androidx.recyclerview.widget.RecyclerView;

import ch.wiss.m335.quiz.data.QuestionEntity;
import ch.wiss.m335.quiz.data.QuizDatabase;
import com.google.android.material.floatingactionbutton.FloatingActionButton;

import java.util.List;

/**
 * Zeigt alle Fragen der Datenbank und führt ins Formular.
 *
 * Der runde Knopf öffnet das Formular ohne Extra, ein Klick auf eine Zeile
 * mit der id der Frage. Beides ist derselbe Intent, darum gibt es nur einen
 * Bildschirm zum Anlegen und Bearbeiten.
 *
 * Der Bestand wird in onResume neu geladen. So steht nach dem Ändern und
 * nach dem Löschen sofort der neue Stand da.
 */
public class AdminActivity extends AppCompatActivity implements QuestionAdapter.OnQuestionClickListener {

    /** Log-Tag für das ganze Modul 335. Derselbe wie in den anderen Activities. */
    private static final String TAG = "Modul335";

    /** Der Titel oben. Er trägt die Anzahl Fragen. */
    private TextView tvAdminTitle;

    /** Der Text, der erscheint, wenn keine Frage da ist. */
    private TextView tvEmpty;

    /** Die Liste der Fragen. */
    private RecyclerView rvQuestions;

    /** Der Adapter, der die Zeilen füllt. */
    private QuestionAdapter adapter;

    /** Der runde Knopf unten rechts. Er öffnet das leere Formular. */
    private FloatingActionButton fabAddQuestion;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_admin);

        // 16dp eigener Rand, in Pixel umgerechnet über die Dichte des Geräts.
        final int extraPadding = Math.round(16 * getResources().getDisplayMetrics().density);

        // Wichtig: der eigene Rand wird zu den System-Insets dazugerechnet.
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.clAdmin), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(
                    systemBars.left + extraPadding,
                    systemBars.top + extraPadding,
                    systemBars.right + extraPadding,
                    systemBars.bottom + extraPadding);
            return insets;
        });

        tvAdminTitle = findViewById(R.id.tvAdminTitle);
        tvEmpty = findViewById(R.id.tvEmpty);
        rvQuestions = findViewById(R.id.rvQuestions);
        fabAddQuestion = findViewById(R.id.fabAddQuestion);

        // Ohne LayoutManager bleibt die Liste leer. Keine Exception, kein
        // Eintrag unter unserem Tag, nur eine leere Fläche.
        rvQuestions.setLayoutManager(new LinearLayoutManager(this));

        // Ohne Extra im Intent. Das Formular weiss damit, dass es eine neue
        // Frage anlegen soll. In 08C kommt hier eine id dazu.
        fabAddQuestion.setOnClickListener(v ->
                startActivity(new Intent(this, QuestionEditActivity.class)));

        adapter = new QuestionAdapter(this);
        rvQuestions.setAdapter(adapter);

        Log.d(TAG, "AdminActivity onCreate");
    }

    /**
     * Reagiert auf einen Klick in der Liste und öffnet die Frage.
     *
     * Es ist derselbe Intent wie beim Plus-Knopf, nur mit der id dazu. Genau
     * darum ist das Formular ein Bildschirm und nicht zwei. Ein zweiter
     * Bildschirm mit denselben elf Feldern müsste bei jeder Änderung zweimal
     * angepasst werden.
     *
     * @param question die angetippte Frage
     */
    @Override
    public void onQuestionClick(QuestionEntity question) {
        Log.d(TAG, "question clicked, id=" + question.getId());

        Intent intent = new Intent(this, QuestionEditActivity.class);
        intent.putExtra(EXTRA_QUESTION_ID, question.getId());
        startActivity(intent);
    }

    /**
     * Schaltet zwischen Liste und leerem Zustand um.
     *
     * Es ist immer genau eines von beiden zu sehen. Darum steht das
     * Umschalten in einer einzigen Methode und nicht an zwei Stellen.
     *
     * @param empty true, wenn keine Frage da ist
     */
    private void showEmptyState(boolean empty) {
        tvEmpty.setVisibility(empty ? View.VISIBLE : View.GONE);
        rvQuestions.setVisibility(empty ? View.GONE : View.VISIBLE);
    }

    /**
     * Lädt die Fragen jedes Mal neu, wenn der Bildschirm erscheint.
     *
     * Dieselbe Überlegung wie bei der Rangliste: onCreate läuft nur
     * einmal, onResume bei jeder Rückkehr. Ab 08B wird das wichtig, denn
     * dann kommt man von einem Formular hierher zurück.
     */
    @Override
    protected void onResume() {
        super.onResume();

        List<QuestionEntity> questions =
                QuizDatabase.getInstance(this).questionDao().getAll();

        adapter.setQuestions(questions);
        tvAdminTitle.setText(getString(R.string.title_admin_format, questions.size()));
        showEmptyState(questions.isEmpty());

        Log.d(TAG, "questions loaded, total=" + questions.size());
    }
}

package ch.wiss.m335.quiz;

import android.os.Bundle;
import android.util.Log;
import android.view.View;
import android.widget.ArrayAdapter;
import android.widget.Button;
import android.widget.RadioGroup;
import android.widget.Spinner;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AlertDialog;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;

import ch.wiss.m335.quiz.data.QuestionEntity;
import ch.wiss.m335.quiz.data.QuizDatabase;
import ch.wiss.m335.quiz.model.QuestionBank;
import com.google.android.material.textfield.TextInputEditText;
import com.google.android.material.textfield.TextInputLayout;

import java.util.Arrays;
import java.util.List;

/**
 * Das Formular für eine Frage. Es legt neue an und ändert bestehende.
 *
 * Welcher der zwei Fälle gilt, hängt allein am Feld questionId. Es kommt
 * aus dem Intent. Fehlt dort ein Extra, bleibt es auf 0, und 0 kann keine
 * echte Frage meinen, weil Room seine ids ab 1 vergibt.
 *
 * Die Unterscheidung wirkt sich an drei Stellen aus. showMode stellt Titel
 * und Löschen-Knopf. loadQuestion füllt die Felder, aber nur beim
 * Bearbeiten. Und onSaveClicked ruft update statt insert.
 */
public class QuestionEditActivity extends AppCompatActivity {

    /**
     * Log-Tag für das ganze Modul 335.
     */
    private static final String TAG = "Modul335";

    /**
     * So viele Zeichen muss ein Fragetext mindestens haben.
     */
    private static final int MIN_TEXT_LENGTH = 10;

    private TextInputLayout tilQuestionText;
    private TextInputEditText etQuestionText;
    private TextInputLayout[] answerLayouts;
    private TextInputEditText[] answerFields;
    private RadioGroup rgCorrect;
    private Spinner spnEditCategory;
    private Spinner spnEditDifficulty;

    /** Der Titel oben. Er sagt, welcher der zwei Fälle gerade gilt. */
    private TextView tvEditTitle;

    /** Der Knopf, der die Frage löscht. Nur beim Bearbeiten sichtbar. */
    private Button btnDeleteQuestion;

    /**
     * Die id der Frage, die bearbeitet wird. 0 heisst neue Frage.
     *
     * Room vergibt seine ids ab 1, also kann 0 nie eine echte Frage meinen.
     * Damit braucht es kein zweites Feld, das den Zustand merkt.
     */
    private int questionId = 0;

    /**
     * Die Kategorien in der Reihenfolge, in der sie im Spinner stehen.
     *
     * Als Feld, weil loadQuestion die Position eines Textes darin sucht.
     * Ohne die Liste müsste man sie dem Adapter wieder abfragen.
     */
    private List<String> categories;

    /**
     * Die drei Schwierigkeitsstufen, genau so wie sie in den Daten stehen.
     */
    private static final List<String> DIFFICULTIES = Arrays.asList("leicht", "mittel", "schwer");

    /**
     * Der Schlüssel für die id im Intent.
     *
     * Die Konstante steht hier und nicht in der AdminActivity. Wer den
     * Bildschirm öffnet, muss den Schlüssel kennen. Wer ihn liest, auch.
     * Steht er an einer Stelle, können die zwei nicht auseinanderlaufen.
     */
    public static final String EXTRA_QUESTION_ID = "question_id";

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_question_edit);

        // 16dp eigener Rand, in Pixel umgerechnet über die Dichte des Geräts.
        final int extraPadding = Math.round(16 * getResources().getDisplayMetrics().density);

        // Type.ime() ist die Tastatur. Fünf Eingabefelder auf einem
        // Bildschirm, und ohne diese Zeile legt sie sich über das Feld,
        // in das du gerade tippst.
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.clQuestionEdit), (v, insets) -> {
            Insets systemBars = insets.getInsets(
                    WindowInsetsCompat.Type.systemBars() | WindowInsetsCompat.Type.ime());
            v.setPadding(
                    systemBars.left + extraPadding,
                    systemBars.top + extraPadding,
                    systemBars.right + extraPadding,
                    systemBars.bottom + extraPadding);
            return insets;
        });

        tilQuestionText = findViewById(R.id.tilQuestionText);
        etQuestionText = findViewById(R.id.etQuestionText);
        rgCorrect = findViewById(R.id.rgCorrect);
        spnEditCategory = findViewById(R.id.spnEditCategory);
        spnEditDifficulty = findViewById(R.id.spnEditDifficulty);
        tvEditTitle = findViewById(R.id.tvEditTitle);
        btnDeleteQuestion = findViewById(R.id.btnDeleteQuestion);

        // Die vier Antwortfelder in ein Array. Danach läuft jede Prüfung in
        // einer Schleife statt vier Mal untereinander. Dasselbe Muster wie bei
        // den vier Antwort-Buttons in deiner GameActivity.
        answerLayouts = new TextInputLayout[]{
                findViewById(R.id.tilAnswer1),
                findViewById(R.id.tilAnswer2),
                findViewById(R.id.tilAnswer3),
                findViewById(R.id.tilAnswer4)
        };
        answerFields = new TextInputEditText[]{
                findViewById(R.id.etAnswer1),
                findViewById(R.id.etAnswer2),
                findViewById(R.id.etAnswer3),
                findViewById(R.id.etAnswer4)
        };

        categories = realCategories();

        spnEditCategory.setAdapter(new ArrayAdapter<>(this,
                android.R.layout.simple_spinner_dropdown_item, categories));
        spnEditDifficulty.setAdapter(new ArrayAdapter<>(this,
                android.R.layout.simple_spinner_dropdown_item, DIFFICULTIES));

        findViewById(R.id.btnSaveQuestion).setOnClickListener(v -> onSaveClicked());
        findViewById(R.id.btnCancelQuestion).setOnClickListener(v -> finish());
        btnDeleteQuestion.setOnClickListener(v -> onDeleteClicked());

        // Die id kommt aus dem Intent. Ist keine da, liefert getIntExtra den
        // zweiten Wert zurück, also 0. Damit steht der Fall Anlegen ohne
        // eine einzige Abfrage fest.
        questionId = getIntent().getIntExtra(EXTRA_QUESTION_ID, 0);

        // Die Reihenfolge zählt. Die Adapter müssen stehen, bevor
        // loadQuestion eine Position darin auswählt.
        showMode();
        if (questionId > 0) {
            loadQuestion();
        }

        Log.d(TAG, "QuestionEditActivity onCreate, id= " + questionId);
    }

    /**
     * Liefert die Kategorien ohne den Eintrag Alle.
     * <p>
     * (HIER: warum Alle hier nicht hingehört, ein bis zwei Zeilen)
     *
     * @return die vier echten Kategorien
     */
    private List<String> realCategories() {
        List<String> all = QuestionBank.categories();
        return all.subList(1, all.size());
    }

    /**
     * Liest ein Feld aus, ohne Leerzeichen am Rand.
     *
     * @param field das Eingabefeld
     * @return der Inhalt, nie null
     */
    private String textOf(TextInputEditText field) {
        return field.getText() == null ? "" : field.getText().toString().trim();
    }

    /**
     * Findet heraus, welcher RadioButton gewählt ist.
     *
     * @return 0 bis 3 für die vier Antworten
     */
    private int selectedAnswerIndex() {
        int checkedId = rgCorrect.getCheckedRadioButtonId();

        if (checkedId == R.id.rbCorrect2) {
            return 1;
        }
        if (checkedId == R.id.rbCorrect3) {
            return 2;
        }
        if (checkedId == R.id.rbCorrect4) {
            return 3;
        }
        // Auch der Fall nichts gewählt landet hier. Im Layout ist der erste
        // Knopf vorausgewählt, also kann das nicht vorkommen.
        return 0;
    }

    /**
     * Gibt den RadioButton zurück basierend auf der übergebenen gecheckten ID
     * @param checkedID id
     * @return RadioButton
     */
    private int radioIdOf(int checkedID){
        if(checkedID == 1){
            return R.id.rbCorrect2;
        }

        if(checkedID == 2){
            return R.id.rbCorrect3;
        }

        if(checkedID == 3){
            return R.id.rbCorrect4;
        }

        return R.id.rbCorrect1;
    }

    /**
     * Baut aus den Eingaben eine Frage und schreibt sie in die Datenbank.
     */
    private void onSaveClicked() {

        if (!isFormValid()) {
            Log.d(TAG, "save rejected, form not valid");
            return;
        }

        QuestionEntity question = new QuestionEntity();
        question.setText(textOf(etQuestionText));
        question.setAnswer1(textOf(answerFields[0]));
        question.setAnswer2(textOf(answerFields[1]));
        question.setAnswer3(textOf(answerFields[2]));
        question.setAnswer4(textOf(answerFields[3]));

        // Die richtige Antwort wird als Text gespeichert und nicht als Nummer.
        // So sieht QuestionEntity von aussen gleich aus wie Question, und
        // isCorrect vergleicht weiterhin zwei Texte.
        question.setCorrectAnswer(textOf(answerFields[selectedAnswerIndex()]));

        question.setCategory((String) spnEditCategory.getSelectedItem());
        question.setDifficulty((String) spnEditDifficulty.getSelectedItem());

        // Die eine Verzweigung, um die es in 08C geht. Beim Anlegen bleibt
        // die id auf 0 und Room vergibt eine neue. Beim Bearbeiten wird sie
        // gesetzt, denn ohne id findet update keine Zeile und tut nichts.
        if (questionId > 0) {
            question.setId(questionId);
            QuizDatabase.getInstance(this).questionDao().update(question);
            Log.d(TAG, "question updated, id=" + questionId);
        } else {
            long newId = QuizDatabase.getInstance(this).questionDao().insert(question);
            Log.d(TAG, "question inserted, id=" + newId);
        }

        // finish schliesst nur diesen Bildschirm. Die Liste darunter lebt
        // weiter und kommt wieder zum Vorschein.
        finish();
    }

    /**
     * Prüft alle Felder und zeigt die Fehler dort an, wo sie entstehen.
     *
     * Wichtig ist die Bauart: die Methode setzt valid auf false und prüft
     * weiter. Sie steigt nicht mit return aus. Wer nach dem ersten Fehler
     * aussteigt, zwingt den Benutzer, mehrmals auf Speichern zu tippen, um
     * mehrere Fehler zu sehen.
     *
     * Und jede Regel räumt ihre eigene Meldung wieder weg. Ohne das else
     * mit setError(null) bleibt eine rote Meldung stehen, obwohl das Feld
     * inzwischen stimmt.
     *
     * Seit 08C sind Antwort 3 und 4 optional, und darum hängen zwei neue
     * Regeln daran. Sie liessen sich anhängen, ohne die alten anzufassen.
     * Genau dafür war der Umbau aus 08B da.
     *
     * Eine Sache ist neu: auf Antwort 3 und 4 zeigen zwei Regeln. Das else
     * der einen würde die Meldung der anderen löschen. Darum werden diese
     * zwei Felder einmal geräumt und danach nur noch gesetzt.
     *
     * @return true, wenn alle Eingaben brauchbar sind
     */
    private boolean isFormValid() {
        boolean valid = true;

        if (textOf(etQuestionText).length() < MIN_TEXT_LENGTH) {
            tilQuestionText.setError(getString(R.string.error_question_text));
            valid = false;
        } else {
            tilQuestionText.setError(null);
        }

        // Antwort 1 und 2 sind weiterhin Pflicht. Eine Frage mit nur einer
        // Antwort ist keine Frage.
        for (int i = 0; i < 2; i++) {
            if (textOf(answerFields[i]).isEmpty()) {
                answerLayouts[i].setError(getString(R.string.error_answer_required));
                valid = false;
            } else {
                answerLayouts[i].setError(null);
            }
        }

        // Antwort 3 und 4 sind seit 08C optional. Auf sie zeigen zwei
        // Regeln statt einer, und darum passt das Muster mit dem else
        // hier nicht mehr. Zwei else könnten sich gegenseitig löschen.
        // Also einmal räumen, danach nur noch setzen.
        answerLayouts[2].setError(null);
        answerLayouts[3].setError(null);

        // Neue Regel 1, die Lücke. Antwort 4 ohne Antwort 3 würde im Spiel
        // einen leeren Knopf zwischen zwei gefüllte setzen.
        if (textOf(answerFields[2]).isEmpty() && !textOf(answerFields[3]).isEmpty()) {
            answerLayouts[2].setError(getString(R.string.error_answer_gap));
            valid = false;
        }

        // Neue Regel 2, der Knopf oben. Der Radiobutton lässt sich auf ein
        // leeres Feld stellen. Dann wäre die richtige Antwort ein leerer
        // Text, und im Spiel gäbe es keine richtige Antwort mehr.
        int correct = selectedAnswerIndex();
        if (textOf(answerFields[correct]).isEmpty()) {
            answerLayouts[correct].setError(getString(R.string.error_correct_empty));
            valid = false;
        }

        return valid;
    }

    /**
     * Schreibt die Frage mit der questionId in die Felder des Formulars.
     *
     * Die Gegenrichtung zu onSaveClicked. Dort wandern die Felder in ein
     * Objekt, hier wandert ein Objekt in die Felder.
     *
     * findById liefert null, wenn es die Frage nicht mehr gibt. Dann wird
     * der Bildschirm geschlossen. Ein leeres Formular anzuzeigen wäre
     * falsch, denn update fände beim Speichern keine Zeile.
     *
     * Die zwei Spinner brauchen selectValue. Ein Spinner hat kein setText,
     * er will eine Position. Ohne die zwei Zeilen bliebe er auf dem ersten
     * Eintrag, und das Speichern schriebe die falsche Kategorie zurück.
     */
    private void loadQuestion() {
        QuestionEntity question = QuizDatabase.getInstance(this).questionDao().findById(questionId);

        if (question == null) {
            Log.d(TAG, "question not found, id=" + questionId);
            finish();
            return;
        }

        etQuestionText.setText(question.getText());

        String[] answers = question.getAnswers();
        for (int i = 0; i < answerFields.length; i++) {
            answerFields[i].setText(answers[i]);
        }

        // Vom Text zurück zum Knopf. selectedAnswerIndex geht die andere
        // Richtung, vom Knopf zum Text.
        for (int i = 0; i < answers.length; i++) {
            if (question.getCorrectAnswer().equals(answers[i])) {
                rgCorrect.check(radioIdOf(i));
                break;
            }
        }

        // Ein Spinner nimmt keinen Text entgegen, sondern eine Position.
        // Ohne diese zwei Zeilen bliebe er auf dem ersten Eintrag stehen,
        // und das Speichern schriebe die falsche Kategorie zurück.
        selectValue(spnEditCategory, categories, question.getCategory());
        selectValue(spnEditDifficulty, DIFFICULTIES, question.getDifficulty());

        Log.d(TAG, "question loaded, id=" + questionId);
    }

    /**
     * Stellt den Bildschirm auf Anlegen oder auf Bearbeiten.
     *
     * Eine einzige Methode schaltet beides um, Titel und Löschen-Knopf.
     * Waeren es zwei Stellen, könnte der Titel Frage bearbeiten sagen,
     * während der Löschen-Knopf fehlt. Dasselbe Muster wie showState in
     * der GameActivity.
     */
    private void showMode() {
        boolean editing = questionId > 0;

        tvEditTitle.setText(editing ? R.string.title_question_edit : R.string.title_question_new);
        btnDeleteQuestion.setVisibility(editing ? View.VISIBLE : View.GONE);
    }

    /**
     * Fragt nach und löscht die Frage.
     *
     * Löschen lässt sich nicht rückgängig machen. Darum die Rückfrage.
     * Sie steht bewusst nicht im Dialog selbst, sondern der Dialog fragt und
     * löscht erst, wenn der Benutzer den zweiten Knopf drückt.
     *
     * setCancelable bleibt auf dem Standard true. Wer neben den Dialog
     * tippt, hat damit abgebrochen, und das ist die harmlose Richtung.
     */
    private void onDeleteClicked() {
        new AlertDialog.Builder(this)
                .setTitle(R.string.dialog_delete_title)
                .setMessage(R.string.dialog_delete_message)
                .setNegativeButton(R.string.dialog_delete_keep, null)
                .setPositiveButton(R.string.dialog_delete_confirm, (dialog, which) -> deleteQuestion())
                .show();
    }

    /**
     * Löscht die Frage wirklich und schliesst den Bildschirm.
     *
     * Room löscht über die id. Die anderen Felder darf das Objekt leer
     * lassen, sie werden nicht angeschaut.
     */
    private void deleteQuestion() {
        QuestionEntity question = new QuestionEntity();
        question.setId(questionId);

        QuizDatabase.getInstance(this).questionDao().delete(question);
        Log.d(TAG, "question deleted, id=" + questionId);

        finish();
    }

    /**
     * Stellt einen Spinner auf den Eintrag mit diesem Text.
     *
     * setSelection will eine Position und keinen Text. Die Position steht in
     * derselben Liste, die auch der Adapter bekommen hat, darum indexOf.
     *
     * Steht der Text nicht in der Liste, liefert indexOf minus eins. Dann
     * bleibt der Spinner, wo er ist. Das passiert, wenn in der Datenbank
     * eine Kategorie steht, die es im Programm nicht mehr gibt.
     *
     * @param spinner der Spinner, der umgestellt wird
     * @param values  die Liste, die der Spinner anzeigt
     * @param value   der Text, der ausgewählt werden soll
     */
    private void selectValue(Spinner spinner, List<String> values, String value) {
        int position = values.indexOf(value);

        if (position >= 0) {
            spinner.setSelection(position);
        }
    }
}

package ch.wiss.m335.quiz.data;

import androidx.room.Dao;
import androidx.room.Delete;
import androidx.room.Insert;
import androidx.room.Query;
import androidx.room.Update;

import java.util.List;

/**
 * Der Zugang zur Tabelle questions.
 *
 * Alle vier CRUD-Operationen stehen hier. insert, getAll und count sind seit
 * 08A in Gebrauch, update, delete und findById seit 08C. getByCategory kam
 * dazu, als das Spiel seine Fragen aus der Datenbank zu holen begann.
 */
@Dao
public interface QuestionDao {

    /**
     * Legt eine neue Frage an.
     *
     * Die id im übergebenen Objekt wird nicht beachtet. Room vergibt eine
     * eigene und gibt sie zurück.
     *
     * @param question die neue Frage
     * @return die id, die Room vergeben hat
     */
    @Insert
    long insert(QuestionEntity question);

    /**
     * Liefert alle Fragen, nach Kategorie und dann nach id sortiert.
     *
     * Die Sortierung steht im SQL und nicht in Java. Die Datenbank kann das
     * besser, und die Liste kommt schon fertig geordnet an.
     *
     * @return alle Fragen, nie null
     */
    @Query("SELECT * FROM questions ORDER BY category, id")
    List<QuestionEntity> getAll();

    /**
     * Zählt alle Fragen.
     *
     * Die einzige Methode, die in 07C schon läuft. Sie beantwortet die Frage,
     * ob überhaupt etwas in der Tabelle steht.
     *
     * @return Anzahl Fragen in der Tabelle
     */
    @Query("SELECT COUNT(*) FROM questions")
    int count();

    /**
     * Speichert Änderungen an einer bestehenden Frage.
     *
     * Room erkennt die Frage an ihrer id. Ist die id 0, findet Room keine
     * passende Zeile, und es passiert gar nichts. Das ist der häufigste
     * Fehler beim Bearbeiten.
     *
     * @param question die geänderte Frage, inklusive ihrer id
     */
    @Update
    void update(QuestionEntity question);

    /**
     * Löscht eine Frage.
     *
     * Auch hier zählt nur die id. Die anderen Felder darf Room ignorieren.
     *
     * @param question die zu löschende Frage
     */
    @Delete
    void delete(QuestionEntity question);

    /**
     * Liefert eine einzelne Frage.
     *
     * Wird beim Bearbeiten gebraucht, um das Formular zu füllen.
     *
     * @param id die Nummer der Frage
     * @return die Frage, oder null wenn es sie nicht gibt
     */
    @Query("SELECT * FROM questions WHERE id = :id")
    QuestionEntity findById(int id);

    /**
     * Liefert alle Fragen einer Kategorie.
     *
     * Der Doppelpunkt vor category ist kein Tippfehler. So markiert Room
     * einen Platzhalter, in den der Wert des Parameters eingesetzt wird.
     * Ohne Platzhalter, also mit zusammengesetztem Text, wäre die Abfrage
     * für eine SQL-Injection offen.
     *
     * Die Sortierung ist dieselbe wie in getAll, damit eine Runde nicht
     * anders geordnet ankommt als die Verwaltungsliste.
     *
     * @param category die Kategorie, genau so geschrieben wie in der Spalte
     * @return alle Fragen dieser Kategorie, nie null
     */
    @Query("SELECT * FROM questions WHERE category = :category ORDER BY category, id")
    List<QuestionEntity> getByCategory(String category);
}

        showScore();

        String[] answers = currentQuestion.getAnswers();
        for(int i=0; i < answerButtons.length; i++){
            Button button = answerButtons[i];
            button.setText(answers[i]);
            button.setEnabled(true);
            button.setBackgroundTintList(defaultButtonTint);
            button.setTextColor(defaultButtonTextColors);

            // Ein leeres Antwortfeld heisst: diese Frage hat weniger als
            // vier Möglichkeiten. Der Knopf verschwindet, statt leer
            // dazustehen. VISIBLE muss dabei stehen bleiben, sonst fehlt
            // der Knopf ab der nächsten Frage für immer.
            button.setVisibility(answers[i].isEmpty() ? View.GONE : View.VISIBLE);
        }

        btnNext.setVisibility(View.GONE);


    }

    /**
     * Startet eine neue Runde mit den ersten fünf Fragen des Bestands.
     *
     * Punktestand und Zähler gehen auf null zurück. Danach ist wieder der
     * Fragebereich sichtbar und der Ergebnisbereich verschwunden.
     */
    private void startRound() {
        // Eine neue Runde macht das alte Ergebnis ungültig, bis sie fertig ist.
        setResult(RESULT_CANCELED);

        currentIndex = 0;
        score = 0;

        showQuestion();
        showState(GameState.QUESTION);

        Log.d(TAG, "round started");
    }

    /**
     * Wird aufgerufen, wenn einer der vier Antwort-Buttons angetippt wird.
     *
     * @param view der Button, der angetippt wurde
     */
    private void onAnswerClicked(View view) {
        String givenAnswer = ((Button) view).getText().toString();
        boolean isCorrect = currentQuestion.isCorrect(givenAnswer);

        if(isCorrect){
            score++;
            showScore();
        }

        markAnswers(givenAnswer);
        lockAnswers();
        btnNext.setVisibility(View.VISIBLE);

        Log.d(TAG, "answer=" + givenAnswer + " correct=" + isCorrect);
    }

    /**
     * Schaltet zur nächsten Frage oder beendet die Runde.
     *
     * Erst hochzählen, dann prüfen, dann entscheiden. Eine Liste mit
     * fünf Einträgen hat die Plätze 0 bis 4.
     */
    private void onNextClicked() {
        currentIndex++;
        if(currentIndex < roundQuestions.size()) {
            showQuestion();
        } else {
            showResult();
        }
    }

    /** Blendet den Fragebereich aus und den Ergebnisbereich ein. */
    private void showResult(){
        tvResult.setText(getString(R.string.result_format, score, roundQuestions.size()));
        buildQuestionList();

        showState(GameState.RESULT);

        publishResult();
        // Der beste Wert wird hier gespeichert und nicht im Menü.
        // Hier entsteht das Ergebnis.
        boolean isNewBest = ScoreStorage.saveIfBetter(this, score);

        Log.d(TAG, "round finished, score=" + score + " newBest=" + isNewBest);
    }

    /**
     * Färbt die richtige Antwort grün. War die Antwort falsch, wird der
     * angetippte Button zusätzlich rot.
     *
     * @param givenAnswer die Antwort, die der Spieler angetippt hat
     */
    private void markAnswers(String givenAnswer){
        for(Button button : answerButtons){
            String text = button.getText().toString();
            if(currentQuestion.isCorrect(text)){
                markButton(button, R.color.answer_correct);
            } else if (text.equals(givenAnswer)) {
                markButton(button, R.color.answer_wrong);
            }
        }
    }

    /**
     * Färbt einen einzelnen Button ein.
     *
     * @param button   der Button, der eingefärbt wird
     * @param colorRes die Farbe aus colors.xml
     */
    private void markButton(Button button, int colorRes) {
        int color = ContextCompat.getColor(this, colorRes);

        // Achtung: das heutige Template erzeugt einen MaterialButton.
        // setBackgroundColor ignoriert er stillschweigend. Nichts stürzt ab,
        // es passiert nur einfach nichts. Richtig ist setBackgroundTintList.
        button.setBackgroundTintList(ColorStateList.valueOf(color));

        // Die Textfarbe wird ausdrücklich gesetzt, weil ein gesperrter
        // Button seine Schrift sonst blass zeichnet.
        button.setTextColor(Color.WHITE);
    }

    /**
     * Sperrt alle vier Antwort-Buttons.
     *
     * Ohne diese Sperre tippt jemand zweimal auf dieselbe richtige Antwort
     * und bekommt zwei Punkte für eine Frage.
     */
    private void lockAnswers(){
        for(Button button : answerButtons){
            button.setEnabled(false);
        }
    }

    /**
     * Hinterlegt das Ergebnis für die MainActivity.
     *
     * Ausgeliefert wird es erst, wenn diese Activity endet.
     */
    private void publishResult() {
        Intent data = new Intent();
        data.putExtra(EXTRA_SCORE, score);
        data.putExtra(EXTRA_TOTAL, roundQuestions.size());
        setResult(RESULT_OK, data);

        Log.d(TAG, "result published, score=" + score + "/" + roundQuestions.size());
    }

    /**
     * Schaltet den Bildschirm auf einen Zustand um.
     *
     * Das ist die einzige Stelle, die über Sichtbarkeit entscheidet.
     */
    private void showState(GameState state) {
        llCategory.setVisibility(state == GameState.CHOOSE_CATEGORY ? View.VISIBLE : View.GONE);
        svQuestion.setVisibility(state == GameState.QUESTION ? View.VISIBLE : View.GONE);
        svResult.setVisibility(state == GameState.RESULT ? View.VISIBLE : View.GONE);

        Log.d(TAG, "showState " + state);
    }

    /**
     * Stellt die Runde aus der gewählten Kategorie zusammen und startet sie.
     *
     * Gemischt wird eine Kopie des Bestands, nicht der Bestand selbst.
     */
    private void onStartClicked() {
        selectedCategory = (String) spnCategory.getSelectedItem();

        // Die Fragen kommen bei jedem Start frisch aus der Datenbank. Wer
        // im Verwaltungsbildschirm eine Frage anlegt, ändert oder löscht,
        // spielt sie in der nächsten Runde. Ein Feld, das den Bestand
        // zwischenspeichert, wäre sofort veraltet.
        QuestionDao dao = QuizDatabase.getInstance(this).questionDao();

        if (QuestionBank.CATEGORY_ALL.equals(selectedCategory)) {
            categoryQuestions = dao.getAll();
        } else {
            categoryQuestions = dao.getByCategory(selectedCategory);
        }

        // Rücksetzen der Felder damit nach der 2. Runde nicht der Name der Vorrunde steht
        scoreSaved = false;
        etPlayerName.setText("");
        updateSavedArea();

        List<QuestionEntity> pool = new ArrayList<>(categoryQuestions);
        Collections.shuffle(pool);

        int count = Math.min(QUESTIONS_PER_ROUND, pool.size());
        roundQuestions = new ArrayList<>(pool.subList(0, count));

        Log.d(TAG, "category=" + selectedCategory + ", bank=" + categoryQuestions.size()
                + ", round=" + count);

        startRound();
    }

    private String currentName() {
        return etPlayerName.getText() == null ? "" : etPlayerName.getText().toString().trim();
    }

    private boolean isNameValid() {
        String name = currentName();

        if(name.isEmpty()) {
            tilPlayerName.setError(getString(R.string.error_name_required));
            return false;
        }
        if(name.length() < MIN_NAME_LENGTH) {
            tilPlayerName.setError(getString(R.string.error_name_too_short));
            return false;
        }

        tilPlayerName.setError(null);
        return true;
    }

    private void onSaveScoreClicked() {
        if(scoreSaved){
            return;
        }

        if(!isNameValid()){
            return;
        }

        String name = currentName();
        HighscoreStorage.add(this, new HighscoreEntry(name, score, selectedCategory));

        scoreSaved = true;
        updateSavedArea();

        Log.d(TAG, "highscore saved for " + name);
    }

    private void updateSavedArea(){
        btnSaveScore.setEnabled(!scoreSaved);
        etPlayerName.setEnabled(!scoreSaved);
        tvSaved.setVisibility(scoreSaved ? View.VISIBLE : View.GONE);
        if(scoreSaved){
            tilPlayerName.setError(null);
        }
    }
}
